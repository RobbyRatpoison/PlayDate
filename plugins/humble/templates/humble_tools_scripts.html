function humbleDownloadPoller(appid) { var iv = setInterval(function() { fetch('/api/humble/download-status/' + appid) .then(function(r) { return r.json(); }) .then(function(d) { if (!d.status) { clearInterval(iv); return; } if (d.status === 'downloading') { var msg = 'Downloading ' + escHtml(d.filename || '') + '…'; if (d.total > 0) { var pct = Math.round(d.received / d.total * 100); msg += ' ' + pct + '%'; } showLaunchToast(msg); } else if (d.status === 'extracting') { showLaunchToast('Extracting ' + escHtml(d.filename || '') + '…'); } else if (d.status === 'done') { clearInterval(iv); showLaunchToast('Downloaded to ~/Games/Humble — ready to install.'); } else if (d.status === 'error') { clearInterval(iv); showLaunchToast('Download failed: ' + escHtml(d.error || 'unknown error')); } }) .catch(function() { clearInterval(iv); }); }, 1000); } function humbleSync() { var statusEl = document.getElementById('humble-manage-status-main'); if (statusEl) statusEl.textContent = 'Starting sync…'; fetch('/api/humble/sync', { method: 'POST' }) .then(function(r) { return r.json(); }) .then(function(data) { if (data.error) { if (statusEl) statusEl.textContent = 'Error: ' + data.error; return; } _humblePollSync(statusEl); }) .catch(function(e) { if (statusEl) statusEl.textContent = 'Error: ' + e.message; }); } function _humblePollSync(statusEl) { var iv = setInterval(function() { fetch('/api/humble/sync-status') .then(function(r) { return r.json(); }) .then(function(d) { if (statusEl) statusEl.textContent = d.status || ''; if (!d.running) { clearInterval(iv); if (d.error && statusEl) statusEl.textContent = 'Error: ' + d.error; } }) .catch(function() { clearInterval(iv); }); }, 1500); }
