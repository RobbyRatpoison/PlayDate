function indiegalaSync() { var statusEl = document.getElementById('indiegala-manage-status-main'); if (statusEl) statusEl.textContent = 'Starting sync…'; fetch('/api/indiegala/sync', { method: 'POST' }) .then(function(r) { return r.json(); }) .then(function(data) { if (data.error) { if (statusEl) statusEl.textContent = 'Error: ' + data.error; return; } _indiegalaPoll(statusEl); }) .catch(function(e) { if (statusEl) statusEl.textContent = 'Error: ' + e.message; }); } function _indiegalaPoll(statusEl) { var iv = setInterval(function() { fetch('/api/indiegala/sync-status') .then(function(r) { return r.json(); }) .then(function(d) { if (statusEl) statusEl.textContent = d.status || ''; if (!d.running) { clearInterval(iv); if (d.error && statusEl) statusEl.textContent = 'Error: ' + d.error; } }) .catch(function() { clearInterval(iv); }); }, 1500); }
