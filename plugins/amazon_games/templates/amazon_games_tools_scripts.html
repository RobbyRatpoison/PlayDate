function amazonGamesSync() { var statusEl = document.getElementById('amazon_games-manage-status-main'); if (statusEl) statusEl.textContent = 'Starting sync…'; fetch('/api/amazon_games/sync', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({source: 'auto'}) }) .then(function(r) { return r.json(); }) .then(function(data) { if (data.error) { if (statusEl) statusEl.textContent = 'Error: ' + data.error; return; } _amazonGamesPoll(statusEl); }) .catch(function(e) { if (statusEl) statusEl.textContent = 'Error: ' + e.message; }); } function _amazonGamesPoll(statusEl) { var iv = setInterval(function() { fetch('/api/amazon_games/sync-status') .then(function(r) { return r.json(); }) .then(function(d) { if (statusEl) statusEl.textContent = d.status || ''; if (!d.running) { clearInterval(iv); if (d.error && statusEl) statusEl.textContent = 'Error: ' + d.error; } }) .catch(function() { clearInterval(iv); }); }, 1500); }
